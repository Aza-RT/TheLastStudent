cmake_minimum_required(VERSION 3.24)
project(TheLastStudent VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(FETCHCONTENT_BASE_DIR "${CMAKE_SOURCE_DIR}/native/third_party/cache" CACHE PATH "Dependencies" FORCE)
include(FetchContent)
option(TLS_BUILD_EXTENSION "Build Godot 4.7.2 extension" ON)
option(TLS_BUILD_TESTBENCH "Build standalone OpenGL testbench" ON)
set(FETCHCONTENT_BASE_DIR "${CMAKE_SOURCE_DIR}/native/third_party/cache" CACHE PATH "Dependencies")
FetchContent_Declare(glm URL https://codeload.github.com/g-truc/glm/tar.gz/refs/tags/1.0.1 URL_HASH SHA256=9f3174561fd26904b23f0db5e560971cbf9b3cbda0b280f04d5c379d03bf234c)
FetchContent_Declare(glew URL https://github.com/Perlmint/glew-cmake/archive/refs/tags/glew-cmake-2.2.0.tar.gz URL_HASH SHA256=cdd82afba80f7cf34548cf5c902240d6721ec5a27ec9e075851b4a3a79ec6907)
set(glew-cmake_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ONLY_LIBS ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(glm glew)
FetchContent_Declare(stb URL https://codeload.github.com/nothings/stb/tar.gz/f0569113c93ad095470c54bf34a17b36646bbbb5 URL_HASH SHA256=4d05c96640ae3a8cbdafdad8d344b50ce610802f78aee80154acdb8e266282e0)
FetchContent_MakeAvailable(stb)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
set(shader_header "#pragma once\nnamespace tls::shaders {\n")
foreach(shader sprite.vert sprite.frag light.vert light.frag fullscreen.vert post.frag)
  file(READ "${CMAKE_SOURCE_DIR}/native/shaders/${shader}" source)
  string(REPLACE "." "_" name "${shader}")
  string(APPEND shader_header "inline constexpr const char* ${name} = R\"GLSL(${source})GLSL\";\n")
endforeach()
string(APPEND shader_header "}\n")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/generated/shaders.hpp" CONTENT "${shader_header}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS native/shaders/sprite.vert native/shaders/sprite.frag native/shaders/light.vert native/shaders/light.frag native/shaders/fullscreen.vert native/shaders/post.frag)
add_library(gfxcore STATIC native/gfxcore/src/renderer.cpp native/gfxcore/src/visibility.cpp native/gfxcore/src/shade_ai.cpp)
target_include_directories(gfxcore PUBLIC native/gfxcore/include PRIVATE "${stb_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(gfxcore PUBLIC glm::glm libglew_static)
target_compile_definitions(gfxcore PUBLIC GLEW_STATIC NOMINMAX)
if(MSVC)
  target_compile_options(gfxcore PRIVATE /W4 /EHsc)
endif()
if(TLS_BUILD_TESTBENCH)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(glfw URL https://codeload.github.com/glfw/glfw/tar.gz/refs/tags/3.4 URL_HASH SHA256=c038d34200234d071fae9345bc455e4a8f2f544ab60150765d7704e08f3dac01)
  FetchContent_MakeAvailable(glfw)
  add_executable(gl_testbench native/testbench/main.cpp)
  target_link_libraries(gl_testbench PRIVATE gfxcore glfw)
  target_compile_definitions(gl_testbench PRIVATE
    TLS_ASSET_DIR="${CMAKE_SOURCE_DIR}/game/assets/art"
    TLS_TESTBENCH_ASSET_DIR="${CMAKE_SOURCE_DIR}/native/testbench/assets"
    TLS_SHADER_DIR="${CMAKE_SOURCE_DIR}/native/shaders"
  )
endif()
if(TLS_BUILD_EXTENSION)
  set(GODOTCPP_CUSTOM_API_FILE "${CMAKE_SOURCE_DIR}/native/extension/extension_api.json" CACHE FILEPATH "Exact Godot 4.7.2 API" FORCE)
  set(GODOTCPP_TARGET template_debug CACHE STRING "" FORCE)
  set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "" FORCE)
  set(GODOTCPP_BUILD_PROFILE "${CMAKE_SOURCE_DIR}/native/extension/build_profile.json" CACHE FILEPATH "" FORCE)
  FetchContent_Declare(godotcpp URL https://codeload.github.com/godotengine/godot-cpp/tar.gz/2bb1acf6f8ae1ce96ac7b9ba3a10b499c59c49ec URL_HASH SHA256=63349d818e612530a2358844d4b0f509b61397fef5169dec9eec1cab1e433202 SOURCE_DIR "${CMAKE_SOURCE_DIR}/native/third_party/godot-cpp")
  FetchContent_MakeAvailable(godotcpp)
  add_library(tls_extension SHARED native/extension/tls_extension.cpp)
  target_link_libraries(tls_extension PRIVATE gfxcore godot-cpp)
  set_target_properties(tls_extension PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/game/bin" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/game/bin")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(tls_extension PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/game/bin" LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/game/bin")
  endforeach()
endif()
enable_testing()
FetchContent_Declare(doctest URL https://codeload.github.com/doctest/doctest/tar.gz/refs/tags/v2.4.11 URL_HASH SHA256=632ed2c05a7f53fa961381497bf8069093f0d6628c5f26286161fbd32a560186)
FetchContent_MakeAvailable(doctest)
add_executable(tls_tests native/tests/tests.cpp native/tests/azat_visibility_cases.cpp)
target_link_libraries(tls_tests PRIVATE gfxcore doctest::doctest)
add_test(NAME native_math_visibility_ai COMMAND tls_tests)
